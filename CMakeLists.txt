option(CYBERATOR_BUILD_VST3 "Build the development Windows x64 JUCE VST3" OFF)
if(CYBERATOR_BUILD_VST3)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The only approved plugin target is Windows x64 VST3")
    endif()
    include(FetchContent)
    FetchContent_Declare(JUCE
        GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
        GIT_TAG 654bdbc7a9db0dae90849b44abe91f339e9f0768
        GIT_SHALLOW FALSE)
    FetchContent_MakeAvailable(JUCE)
    juce_add_plugin(Cyberator
        COMPANY_NAME "125A Systems"
        BUNDLE_ID "com.125asystems.cyberator"
        PLUGIN_MANUFACTURER_CODE A125
        PLUGIN_CODE Cybr
        FORMATS VST3
        PRODUCT_NAME "125A CYBERATOR"
        IS_SYNTH FALSE
        NEEDS_MIDI_INPUT FALSE
        NEEDS_MIDI_OUTPUT FALSE
        COPY_PLUGIN_AFTER_BUILD FALSE)
    target_sources(Cyberator PRIVATE src/plugin/Processor.cpp)
    target_include_directories(Cyberator PRIVATE src)
    target_compile_features(Cyberator PRIVATE cxx_std_23)
    target_compile_definitions(Cyberator PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
    target_link_libraries(Cyberator PRIVATE juce::juce_audio_processors juce::juce_recommended_config_flags)
    if(BUILD_TESTING)
        add_executable(native_processor_test tests/native_processor_test.cpp)
        target_include_directories(native_processor_test PRIVATE src)
        target_compile_features(native_processor_test PRIVATE cxx_std_23)
        target_compile_definitions(native_processor_test PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
        target_link_libraries(native_processor_test PRIVATE Cyberator juce::juce_audio_processors juce::juce_recommended_config_flags)
        add_test(NAME native_processor_smoke COMMAND native_processor_test)
        add_executable(vst3_host_test tests/vst3_host_test.cpp)
        target_compile_features(vst3_host_test PRIVATE cxx_std_23)
        target_compile_definitions(vst3_host_test PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
        target_link_libraries(vst3_host_test PRIVATE juce::juce_audio_processors juce::juce_recommended_config_flags)
        add_dependencies(vst3_host_test Cyberator_VST3)
        add_test(NAME vst3_host_smoke COMMAND vst3_host_test $<TARGET_FILE:Cyberator_VST3>)
    endif()
endif()
